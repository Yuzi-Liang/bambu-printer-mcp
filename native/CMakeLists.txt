cmake_minimum_required(VERSION 3.20)
project(bambu_windows_helpers LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Bambu Studio Windows helpers require 64-bit Windows and MSVC.")
endif()

add_executable(bambu-print-history bambu-print-history.cpp)
add_executable(bambu-native-print bambu-native-print.cpp)
add_executable(bambu-cad-sandbox bambu-cad-sandbox.cpp)
target_link_libraries(bambu-cad-sandbox PRIVATE advapi32 userenv)
foreach(helper IN ITEMS bambu-print-history bambu-native-print bambu-cad-sandbox)
  target_compile_features(${helper} PRIVATE cxx_std_17)
  set_property(TARGET ${helper} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endforeach()
